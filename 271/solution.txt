class Solution {
public:
    string encode(vector<string>& arr) {
        string ans = "";

        for(int i = 0; i < arr.size(); i++) {
            ans += to_string(arr[i].length()) + "/:" + arr[i];
        }

        return ans;
    }

    vector<string> decode(string &s) {
        vector<string> ans;
        int i = 0;

        while(i < s.length()) {
            string num = "";

            while(s[i] != '/') {
                num += s[i];
                i++;
            }

            int len = stoi(num);
            i += 2;

            string temp = "";

            for(int j = 0; j < len; j++) {
                temp += s[i];
                i++;
            }

            ans.push_back(temp);
        }

        return ans;
    }
};
